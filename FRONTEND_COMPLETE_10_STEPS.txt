cat > src/hooks/use-size-picker.ts << 'EOF'
import { useState, useCallback, useMemo } from 'react';

export interface Size {
  id: string;
  name: string;
  available: boolean;
}

export function usePickerSize({
  initialSizeId = null,
  sizes,
}: {
  initialSizeId?: string | null;
  sizes: Size[];
}) {
  const [selectedSizeId, setSelectedSizeId] = useState<string | null>(initialSizeId);

  const selectedSize = useMemo(
    () => sizes.find((s) => s.id === selectedSizeId) || null,
    [selectedSizeId, sizes]
  );

  const canSelectSize = useCallback((sizeId: string) => {
    return sizes.find((s) => s.id === sizeId)?.available ?? false;
  }, [sizes]);

  const selectSize = useCallback((sizeId: string) => {
    if (canSelectSize(sizeId)) {
      setSelectedSizeId(sizeId);
    }
  }, [canSelectSize]);

  return {
    selectedSizeId,
    selectedSize,
    selectSize,
    canSelectSize,
  };
}
EOF

  onSearch,

COMMAND 6.1: Document Redux Patterns

cat > src/docs/REDUX_PATTERNS.md << 'EOF'
# Redux Patterns Guide

## Store Structure
```
src/redux/
├── store.ts
├── slices/
│   ├── auth.slice.ts
│   ├── product.slice.ts
│   ├── cart.slice.ts
│   └── ...
└── selectors/
    ├── auth.selectors.ts
    ├── product.selectors.ts
    └── ...
```

## Slice Pattern
Each slice should include:
- Initial state (typed)
- Reducers (synchronous actions)
- Extra reducers (async thunk handlers)
- Selectors (in separate file)

## Selector Pattern
Use createSelector for computed values:
```typescript
export const selectCartTotal = createSelector(
  selectCartItems,
  selectTaxRate,
  (items, taxRate) => items.reduce((sum, item) => sum + item.price, 0) * (1 + taxRate)
);
```

## Async Thunk Pattern
```typescript
export const fetchProducts = createAsyncThunk(
  'products/fetchProducts',
  async (params: FetchParams) => {
    const response = await api.get(PRODUCT_ENDPOINTS.GET_ALL);
    return response.data;
  }
);
```

## Checklist
- [ ] All slices use createSlice
- [ ] All async operations use createAsyncThunk
- [ ] All selectors use createSelector
- [ ] No hardcoded strings in reducer names
- [ ] State is normalized where applicable
EOF

COMMAND 9.1: Create Verification Script

cat > src/scripts/verify-refactoring.sh << 'EOF'
#!/bin/bash

GREEN='\033[0;32m'
RED='\033[0;31m'
NC='\033[0m'

echo "=== FRONTEND REFACTORING VERIFICATION ==="
echo ""

# Test 1: Build
echo -n "Build... "
if npm run build > /dev/null 2>&1; then
  echo -e "${GREEN}✓${NC}"
else
  echo -e "${RED}✗${NC}"
  exit 1
fi

# Test 2: TypeScript
echo -n "TypeScript... "
if npx tsc --noEmit 2> /dev/null; then
  echo -e "${GREEN}✓${NC}"
else
  echo -e "${RED}✗${NC}"
fi

# Test 3: Components Count
COMPONENTS=$(find src/components -name "*.tsx" | wc -l)
echo "Components: ${GREEN}${COMPONENTS}${NC}"

# Test 4: Console.log Check
CONSOLE=$(grep -r "console\.log" src --include="*.tsx" --include="*.ts" | grep -v ".backup" | wc -l)
if [ $CONSOLE -eq 0 ]; then
  echo -e "Console.log: ${GREEN}0 ✓${NC}"
else
  echo -e "Console.log: ${RED}${CONSOLE}${NC}"
fi

# Test 5: Component Sizes
LARGE=$(find src/components -name "*.tsx" -exec wc -l {} \; | awk '$1 > 400' | wc -l)
echo "Large components (>400 lines): ${GREEN}${LARGE}${NC}"

# Test 6: Constants Count
CONSTANTS=$(find src/constants -type f -name "*.ts" | wc -l)
echo -e "Constants files: ${GREEN}${CONSTANTS}${NC}"

# Test 7: Hooks Count
HOOKS=$(find src/hooks -type f -name "*.ts" | wc -l)
echo -e "Hooks files: ${GREEN}${HOOKS}${NC}"

echo ""
echo "=== VERIFICATION COMPLETE ==="
EOF

chmod +x src/scripts/verify-refactoring.sh
