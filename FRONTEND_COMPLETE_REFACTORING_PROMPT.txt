"use client";
import { useMemo } from "react";
import { Badge } from "@/components/ui/badge";
import { cn } from "@/lib/utils";
import { ProductSize } from "@/redux/features/business/store/models/response/product-response";
import { PRODUCT_STRINGS } from "@/constants/ui-strings";

interface SizeGridProps {
  sizes: ProductSize[] | undefined;
  selectedSize: ProductSize | null;
  onSelectSize: (size: ProductSize) => void;
  quantityMap?: Map<string, number>;
}

export function SizeGrid({
  sizes,
  selectedSize,
  onSelectSize,
  quantityMap = new Map(),
}: SizeGridProps) {
  if (!sizes || sizes.length === 0) return null;

  return (
    <div className="space-y-3">
      <label className="text-sm font-medium">{PRODUCT_STRINGS.SELECT_SIZE}</label>
      <div className="grid grid-cols-3 gap-2 sm:grid-cols-4">
        {sizes.map((size) => {
          const hasStock = (size.stock ?? 0) > 0;
          const quantity = quantityMap.get(size.id) ?? 0;
          const isSelected = selectedSize?.id === size.id;

          return (
            <button
              key={size.id}
              onClick={() => onSelectSize(size)}
              disabled={!hasStock}
              className={cn(
                "relative p-2 text-center rounded-lg border-2 transition-all",
                isSelected ? "border-primary bg-primary/10" : "border-border hover:border-primary/50",
                !hasStock && "opacity-50 cursor-not-allowed",
              )}
            >
              <div className="font-medium text-sm">{size.name}</div>
              {hasStock && quantity > 0 && (
                <Badge variant="secondary" className="absolute -top-2 -right-2 text-xs">
                  {quantity}
                </Badge>
              )}
              {!hasStock && <div className="text-xs text-muted-foreground">{PRODUCT_STRINGS.OUT_OF_STOCK}</div>}
            </button>
          );
        })}
      </div>
    </div>
  );
}

"use client";
import { Checkbox } from "@/components/ui/checkbox";
import { cn } from "@/lib/utils";

export interface Customization {
  id: string;
  name: string;
  description?: string;
}

interface CustomizationSelectorProps {
  customizations: Customization[] | undefined;
  selectedIds: Set<string>;
  onToggle: (id: string) => void;
  disabled?: boolean;
}

export function CustomizationSelector({
  customizations,
  selectedIds,
  onToggle,
  disabled = false,
}: CustomizationSelectorProps) {
  if (!customizations || customizations.length === 0) return null;

  return (
    <div className="space-y-2">
      <label className="text-sm font-medium">Customizations</label>
      <div className="space-y-2">
        {customizations.map((custom) => (
          <div
            key={custom.id}
            className={cn(
              "flex items-start space-x-2 p-2 rounded border",
              disabled && "opacity-50 cursor-not-allowed",
            )}
          >
            <Checkbox
              id={custom.id}
              checked={selectedIds.has(custom.id)}
              onCheckedChange={() => !disabled && onToggle(custom.id)}
              disabled={disabled}
            />
            <label htmlFor={custom.id} className={cn("text-sm", disabled && "cursor-not-allowed")}>
              <div className="font-medium">{custom.name}</div>
              {custom.description && <div className="text-xs text-muted-foreground">{custom.description}</div>}
            </label>
          </div>
        ))}
      </div>
    </div>
  );
}

"use client";
import { formatCurrency } from "@/utils/common/currency-format";
import { cn } from "@/lib/utils";

interface PriceDisplayProps {
  currentPrice: number;
  originalPrice?: number;
  size?: string;
  quantity?: number;
  showTotal?: boolean;
}

export function PriceDisplay({
  currentPrice,
  originalPrice,
  size,
  quantity = 1,
  showTotal = true,
}: PriceDisplayProps) {
  const hasDiscount = originalPrice && originalPrice > currentPrice;
  const totalPrice = currentPrice * quantity;

  return (
    <div className="space-y-2 py-3 border-t">
      {size && <div className="text-sm text-muted-foreground">Size: {size}</div>}

      <div className="flex items-center justify-between">
        <span className="text-sm">Price per item:</span>
        <div className="flex items-center gap-2">
          {hasDiscount && <span className="text-xs line-through text-muted-foreground">{formatCurrency(originalPrice)}</span>}
          <span className={cn("font-medium", hasDiscount && "text-green-600")}>
            {formatCurrency(currentPrice)}
          </span>
        </div>
      </div>

      {showTotal && quantity > 1 && (
        <div className="flex items-center justify-between text-sm font-medium">
          <span>Total ({quantity}x):</span>
          <span>{formatCurrency(totalPrice)}</span>
        </div>
      )}
    </div>
  );
}
